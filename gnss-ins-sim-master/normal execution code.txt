"""
IMU Data Generator Executable
Generates simulated IMU data from trajectory definition files.
"""

import os
import sys
import math
import argparse
from pathlib import Path

# Handle PyInstaller resource path
def resource_path(relative_path):
    """Get absolute path to resource, works for dev and for PyInstaller"""
    try:
        base_path = sys._MEIPASS
    except Exception:
        base_path = os.path.abspath(".")
    return os.path.join(base_path, relative_path)

from gnss_ins_sim.sim import imu_model
from gnss_ins_sim.sim import ins_sim

D2R = math.pi/180

def main():
    parser = argparse.ArgumentParser(
        description='GNSS-INS-SIM: Generate simulated IMU data from trajectory files',
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  %(prog)s -t demo_motion_def_files/motion_def-static.csv
  %(prog)s -t my_trajectory.csv -o ./output_data -a mid-accuracy
  %(prog)s -t trajectory.csv --fs 100 --no-plot
        """
    )
    
    # Required arguments
    parser.add_argument('-t', '--trajectory', 
                       required=True,
                       help='Path to trajectory definition CSV file')
    
    # Optional arguments
    parser.add_argument('-o', '--output', 
                       default='./imu_data/',
                       help='Output directory for generated data (default: ./imu_data/)')
    
    parser.add_argument('-a', '--imu-accuracy', 
                       default='jz105-rlg',
                       choices=['low-accuracy', 'mid-accuracy', 'high-accuracy', 'jz105-rlg'],
                       help='IMU accuracy model (default: jz105-rlg)')
    
    parser.add_argument('--fs', 
                       type=float, 
                       default=200.0,
                       help='IMU sample frequency in Hz (default: 200.0)')
    
    parser.add_argument('--fs-gps', 
                       type=float, 
                       default=10.0,
                       help='GPS sample frequency in Hz (default: 10.0)')
    
    parser.add_argument('--no-plot', 
                       action='store_true',
                       help='Disable plotting')
    
    parser.add_argument('--ref-frame', 
                       type=int,
                       default=0,
                       choices=[0, 1],
                       help='Reference frame: 0=NED, 1=Virtual inertial frame (default: 0)')
    
    parser.add_argument('--axis', 
                       type=int,
                       default=6,
                       choices=[6, 9],
                       help='IMU axis: 6=gyro+accel, 9=gyro+accel+mag (default: 6)')
    
    parser.add_argument('--gps', 
                       action='store_true',
                       help='Generate GPS data')
    
    args = parser.parse_args()
    
    # Print header
    print("=" * 70)
    print("                    GNSS-INS-SIM: IMU Data Generator")
    print("=" * 70)
    print(f"Trajectory file:  {args.trajectory}")
    print(f"Output directory: {args.output}")
    print(f"IMU accuracy:     {args.imu_accuracy}")
    print(f"IMU sample rate:  {args.fs} Hz")
    print(f"GPS sample rate:  {args.fs_gps} Hz")
    print(f"Reference frame:  {'NED' if args.ref_frame == 0 else 'Virtual Inertial'}")
    print("=" * 70)
    
    # Validate trajectory file exists
    if not os.path.exists(args.trajectory):
        print(f"\nERROR: Trajectory file not found: {args.trajectory}")
        sys.exit(1)
    
    # Create output directory
    os.makedirs(args.output, exist_ok=True)
    
    # Determine magnetometer sample rate
    fs_mag = None if args.axis == 6 else args.fs
    
    try:
        print("\nInitializing IMU model...")
        # Create IMU object
        imu = imu_model.IMU(
            accuracy=args.imu_accuracy, 
            axis=args.axis, 
            gps=args.gps
        )
        
        print("Creating simulation...")
        # Create simulation object
        sim = ins_sim.Sim(
            [args.fs, args.fs_gps, fs_mag],
            args.trajectory,
            ref_frame=args.ref_frame,
            imu=imu,
            mode=None,
            env=None,
            algorithm=None
        )
        
        print("Running simulation...")
        sim.run(1)
        
        # Get simulation stats
        duration = sim.time[-1] if hasattr(sim, 'time') and len(sim.time) > 0 else 0
        num_samples = len(sim.time) if hasattr(sim, 'time') else 0
        
        print(f"\n✓ Simulation complete!")
        #print(f"  Duration: {duration:.2f} seconds")
        #print(f"  Samples:  {num_samples}")
        
        print(f"\nSaving results to {args.output}...")
        sim.results(args.output)
        
        # Generate simple CSV files for easy access
        save_simple_csv_files(sim, args.output)
        
        # Plot if requested
        if not args.no_plot:
            print("\nGenerating plots...")
            try:
                import matplotlib.pyplot as plt
                sim.plot(['ref_pos', 'gyro', 'accel'], opt={'ref_pos': '3d'})
                plt.show(block=True)
            except Exception as e:
                print(f"Warning: Could not generate plots: {e}")
        
        print("\n" + "=" * 70)
        print("✓ Data generation complete!")
        print(f"  Files saved in: {os.path.abspath(args.output)}")
        print("=" * 70)
        
        # List generated files
        print("\nGenerated files:")
        for file in sorted(os.listdir(args.output)):
            file_path = os.path.join(args.output, file)
            if os.path.isfile(file_path):
                size = os.path.getsize(file_path) / 1024  # KB
                print(f"  - {file} ({size:.1f} KB)")
                
    except Exception as e:
        print(f"\nERROR: {e}")
        import traceback
        traceback.print_exc()
        sys.exit(1)

def save_simple_csv_files(sim, output_dir):
    """Save data in simple CSV format for easy access"""
    import numpy as np
    
    if hasattr(sim, 'time') and len(sim.time) > 0:
        # Save gyroscope data
        if hasattr(sim, 'gyro') and sim.gyro is not None:
            gyro_data = np.column_stack([sim.time, sim.gyro])
            np.savetxt(os.path.join(output_dir, 'gyro.csv'),
                      gyro_data,
                      delimiter=',',
                      header='time(s),gyro_x(rad/s),gyro_y(rad/s),gyro_z(rad/s)',
                      comments='',
                      fmt='%.6e')
        
        # Save accelerometer data
        if hasattr(sim, 'accel') and sim.accel is not None:
            accel_data = np.column_stack([sim.time, sim.accel])
            np.savetxt(os.path.join(output_dir, 'accel.csv'),
                      accel_data,
                      delimiter=',',
                      header='time(s),accel_x(m/s^2),accel_y(m/s^2),accel_z(m/s^2)',
                      comments='',
                      fmt='%.6e')
        
        # Save reference position
        if hasattr(sim, 'ref_pos') and sim.ref_pos is not None:
            ref_pos_data = np.column_stack([sim.time, sim.ref_pos])
            np.savetxt(os.path.join(output_dir, 'ref_pos.csv'),
                      ref_pos_data,
                      delimiter=',',
                      header='time(s),lat(rad),lon(rad),alt(m)',
                      comments='',
                      fmt='%.6e')
        
        # Save reference velocity
        if hasattr(sim, 'ref_vel') and sim.ref_vel is not None:
            ref_vel_data = np.column_stack([sim.time, sim.ref_vel])
            np.savetxt(os.path.join(output_dir, 'ref_vel.csv'),
                      ref_vel_data,
                      delimiter=',',
                      header='time(s),vn(m/s),ve(m/s),vd(m/s)',
                      comments='',
                      fmt='%.6e')
        
        # Save reference attitude
        if hasattr(sim, 'ref_att_euler') and sim.ref_att_euler is not None:
            att_data = np.column_stack([sim.time, 
                                       sim.ref_att_euler[:, 0] * 180/math.pi,  # roll to deg
                                       sim.ref_att_euler[:, 1] * 180/math.pi,  # pitch to deg
                                       sim.ref_att_euler[:, 2] * 180/math.pi]) # yaw to deg
            np.savetxt(os.path.join(output_dir, 'ref_att.csv'),
                      att_data,
                      delimiter=',',
                      header='time(s),roll(deg),pitch(deg),yaw(deg)',
                      comments='',
                      fmt='%.6e')

if __name__ == '__main__':
    main()
